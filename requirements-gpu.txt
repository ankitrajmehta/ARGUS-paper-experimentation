# Model inference. Install on the GPU box, on top of requirements.txt.
#
# InternVL2.5-8B needs ~17 GB in fp16 / bf16, or ~6 GB in 4-bit (bitsandbytes).
# It ships custom modelling code, so trust_remote_code=True is required.
# Pin transformers: InternVL2.5's remote code breaks on the 4.5x series.
-r requirements.txt

torch>=2.1
torchvision
# 4.49.x is the only window where both model families work: Qwen2.5-VL was
# added in 4.49, and InternVL2.5's remote code breaks on 4.50+.
transformers>=4.49,<4.50
# Qwen's image preprocessing helper.
qwen-vl-utils
accelerate>=0.26
# Pinned: sentencepiece 0.2.2 rejects InternVL's tokenizer.model with
# "INTERNAL: piece must not include null character", even though the file is
# intact (its sha256 matches the hub). AutoTokenizer swallows that error and
# returns False, so it surfaces much later as "'bool' has no attribute
# convert_tokens_to_ids". 0.2.1 loads it correctly and, unlike 0.1.99, ships
# wheels for Python 3.13 -- which Kaggle now runs.
sentencepiece==0.2.1
# InternVL's tokenizer is sentencepiece-based and loaded with use_fast=False,
# which needs protobuf to parse tokenizer.model.
protobuf
einops
timm
# Only needed for --load-4bit on a <24 GB card.
bitsandbytes; platform_system == "Linux"
